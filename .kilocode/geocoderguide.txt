You'll also need a Mapbox access token. Get one from [Mapbox's website](https://account.mapbox.com/access-tokens/).

## Geocoding Service Integration

Create a geocoding service that can work with various providers:

```jsx
// geocodingService.js
class GeocodingService {
  constructor(provider = 'mapbox', apiKey) {
    this.provider = provider;
    this.apiKey = apiKey;
  }

  async search(query, options = {}) {
    const { limit = 10, proximity, bbox, country } = options;
    
    switch (this.provider) {
      case 'mapbox':
        return this.searchMapbox(query, { limit, proximity, bbox, country });
      case 'nominatim':
        return this.searchNominatim(query, { limit });
      default:
        throw new Error(`Unsupported provider: ${this.provider}`);
    }
  }

  async searchMapbox(query, options) {
    const { limit, proximity, bbox, country } = options;
    let url = `https://api.mapbox.com/geocoding/v5/mapbox.places/${encodeURIComponent(query)}.json?access_token=${this.apiKey}&limit=${limit}`;
    
    if (proximity) url += `&proximity=${proximity.join(',')}`;
    if (bbox) url += `&bbox=${bbox.join(',')}`;
    if (country) url += `&country=${country}`;

    try {
      const response = await fetch(url);
      const data = await response.json();
      
      return {
        results: data.features.map(feature => ({
          id: feature.id,
          name: feature.place_name,
          coordinates: feature.center,
          bbox: feature.bbox,
          relevance: feature.relevance,
          context: feature.context,
          properties: feature.properties
        })),
        total: data.features.length
      };
    } catch (error) {
      console.error('Geocoding error:', error);
      return { results: [], total: 0 };
    }
  }

  async searchNominatim(query, options) {
    const { limit } = options;
    const url = `https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(query)}&limit=${limit}&addressdetails=1`;

    try {
      const response = await fetch(url);
      const data = await response.json();
      
      return {
        results: data.map(item => ({
          id: item.place_id,
          name: item.display_name,
          coordinates: [parseFloat(item.lon), parseFloat(item.lat)],
          bbox: item.boundingbox ? [
            parseFloat(item.boundingbox[2]), // west
            parseFloat(item.boundingbox[0]), // south
            parseFloat(item.boundingbox[3]), // east
            parseFloat(item.boundingbox[1])  // north
          ] : null,
          relevance: parseFloat(item.importance || 0),
          properties: {
            type: item.type,
            class: item.class,
            address: item.address
          }
        })),
        total: data.length
      };
    } catch (error) {
      console.error('Geocoding error:', error);
      return { results: [], total: 0 };
    }
  }
}

export default GeocodingService;
```

## Search Bar Component

Create a search bar with debounced input and loading states:

```jsx
// SearchBar.jsx
import React, { useState, useEffect, useCallback } from 'react';

function SearchBar({ onSearch, isLoading, placeholder = "Search for places..." }) {
  const [query, setQuery] = useState('');
  const [debouncedQuery, setDebouncedQuery] = useState('');

  // Debounce search input
  useEffect(() => {
    const timer = setTimeout(() => {
      setDebouncedQuery(query);
    }, 300);

    return () => clearTimeout(timer);
  }, [query]);

  // Trigger search when debounced query changes
  useEffect(() => {
    if (debouncedQuery.trim() && debouncedQuery.length >= 2) {
      onSearch(debouncedQuery);
    }
  }, [debouncedQuery, onSearch]);

  const handleInputChange = (e) => {
    setQuery(e.target.value);
  };

  const handleClear = () => {
    setQuery('');
    setDebouncedQuery('');
  };

  return (
    <div className="search-bar">
      <div className="search-input-container">
        <input
          type="text"
          value={query}
          onChange={handleInputChange}
          placeholder={placeholder}
          className="search-input"
          disabled={isLoading}
        />
        {isLoading && <div className="search-spinner">⏳</div>}
        {query && !isLoading && (
          <button onClick={handleClear} className="search-clear">
            ✕
          </button>
        )}
      </div>
    </div>
  );
}

export default SearchBar;
```

## Results Panel with Pagination

Create a results panel that handles pagination and result selection:

```jsx
// ResultsPanel.jsx
import React from 'react';

function ResultsPanel({ 
  results, 
  currentPage, 
  itemsPerPage, 
  totalResults, 
  onPageChange, 
  onResultClick,
  selectedResult,
  isVisible 
}) {
  const totalPages = Math.ceil(totalResults / itemsPerPage);
  const startIndex = (currentPage - 1) * itemsPerPage;
  const endIndex = startIndex + itemsPerPage;
  const currentResults = results.slice(startIndex, endIndex);

  const generatePageNumbers = () => {
    const pages = [];
    const maxVisiblePages = 5;
    let startPage = Math.max(1, currentPage - Math.floor(maxVisiblePages / 2));
    let endPage = Math.min(totalPages, startPage + maxVisiblePages - 1);
    
    if (endPage - startPage < maxVisiblePages - 1) {
      startPage = Math.max(1, endPage - maxVisiblePages + 1);
    }

    for (let i = startPage; i <= endPage; i++) {
      pages.push(i);
    }
    
    return pages;
  };

  if (!isVisible || results.length === 0) {
    return null;
  }

  return (
    <div className="results-panel">
      <div className="results-header">
        <h3>Search Results</h3>
        <span className="results-count">
          {totalResults} result{totalResults !== 1 ? 's' : ''} found
        </span>
      </div>

      <div className="results-list">
        {currentResults.map((result, index) => (
          <div
            key={result.id}
            className={`result-item ${selectedResult?.id === result.id ? 'selected' : ''}`}
            onClick={() => onResultClick(result)}
          >
            <div className="result-name">{result.name}</div>
            <div className="result-coords">
              {result.coordinates[1].toFixed(4)}, {result.coordinates[0].toFixed(4)}
            </div>
            {result.relevance && (
              <div className="result-relevance">
                Relevance: {(result.relevance * 100).toFixed(0)}%
              </div>
            )}
          </div>
        ))}
      </div>

      {totalPages > 1 && (
        <div className="pagination">
          <button
            onClick={() => onPageChange(currentPage - 1)}
            disabled={currentPage === 1}
            className="pagination-btn"
          >
            Previous
          </button>

          <div className="pagination-numbers">
            {currentPage > 3 && (
              <>
                <button onClick={() => onPageChange(1)} className="pagination-btn">1</button>
                {currentPage > 4 && <span className="pagination-ellipsis">...</span>}
              </>
            )}

            {generatePageNumbers().map(page => (
              <button
                key={page}
                onClick={() => onPageChange(page)}
                className={`pagination-btn ${page === currentPage ? 'active' : ''}`}
              >
                {page}
              </button>
            ))}

            {currentPage < totalPages - 2 && (
              <>
                {currentPage < totalPages - 3 && <span className="pagination-ellipsis">...</span>}
                <button onClick={() => onPageChange(totalPages)} className="pagination-btn">
                  {totalPages}
                </button>
              </>
            )}
          </div>

          <button
            onClick={() => onPageChange(currentPage + 1)}
            disabled={currentPage === totalPages}
            className="pagination-btn"
          >
            Next
          </button>
        </div>
      )}
    </div>
  );
}

export default ResultsPanel;
```

## Complete Implementation

Here's the main component that brings everything together:

```jsx
// GeocodingMap.jsx
import React, { useState, useCallback, useRef } from 'react';
import Map, { Marker, Popup } from 'react-map-gl';
import SearchBar from './SearchBar';
import ResultsPanel from './ResultsPanel';
import GeocodingService from './geocodingService';
import 'mapbox-gl/dist/mapbox-gl.css';
import './GeocodingMap.css';

const MAPBOX_TOKEN = 'your_mapbox_token_here';

function GeocodingMap() {
  const [viewState, setViewState] = useState({
    longitude: -100,
    latitude: 40,
    zoom: 3.5
  });

  const [searchResults, setSearchResults] = useState([]);
  const [selectedResult, setSelectedResult] = useState(null);
  const [currentPage, setCurrentPage] = useState(1);
  const [itemsPerPage] = useState(5);
  const [isLoading, setIsLoading] = useState(false);
  const [showResults, setShowResults] = useState(false);
  const [showPopup, setShowPopup] = useState(false);

  const geocodingService = useRef(new GeocodingService('mapbox', MAPBOX_TOKEN));

  const handleSearch = useCallback(async (query) => {
    setIsLoading(true);
    setShowResults(false);

    try {
      const response = await geocodingService.current.search(query, {
        limit: 50, // Get more results for pagination
        proximity: [viewState.longitude, viewState.latitude]
      });

      setSearchResults(response.results);
      setCurrentPage(1);
      setShowResults(response.results.length > 0);
      setSelectedResult(null);
      setShowPopup(false);
    } catch (error) {
      console.error('Search failed:', error);
      setSearchResults([]);
      setShowResults(false);
    } finally {
      setIsLoading(false);
    }
  }, [viewState.longitude, viewState.latitude]);

  const handleResultClick = useCallback((result) => {
    setSelectedResult(result);
    setShowPopup(true);
    
    // Fly to the selected location
    setViewState(prev => ({
      ...prev,
      longitude: result.coordinates[0],
      latitude: result.coordinates[1],
      zoom: Math.max(prev.zoom, 12)
    }));
  }, []);

  const handlePageChange = useCallback((newPage) => {
    setCurrentPage(newPage);
  }, []);

  return (
    <div className="geocoding-map-container">
      <div className="map-controls">
        <SearchBar
          onSearch={handleSearch}
          isLoading={isLoading}
          placeholder="Search for places, addresses, or coordinates..."
        />
        
        <ResultsPanel
          results={searchResults}
          currentPage={currentPage}
          itemsPerPage={itemsPerPage}
          totalResults={searchResults.length}
          onPageChange={handlePageChange}
          onResultClick={handleResultClick}
          selectedResult={selectedResult}
          isVisible={showResults}
        />
      </div>

      <Map
        {...viewState}
        onMove={evt => setViewState(evt.viewState)}
        style={{ width: '100%', height: '100vh' }}
        mapStyle="mapbox://styles/mapbox/streets-v12"
        mapboxAccessToken={MAPBOX_TOKEN}
      >
        {/* Show markers for all search results */}
        {searchResults.map((result, index) => (
          <Marker
            key={result.id}
            longitude={result.coordinates[0]}
            latitude={result.coordinates[1]}
            onClick={() => handleResultClick(result)}
          >
            <div className={`marker ${selectedResult?.id === result.id ? 'selected' : ''}`}>
              📍
            </div>
          </Marker>
        ))}

        {/* Show popup for selected result */}
        {selectedResult && showPopup && (
          <Popup
            longitude={selectedResult.coordinates[0]}
            latitude={selectedResult.coordinates[1]}
            onClose={() => setShowPopup(false)}
            closeButton={true}
          >
            <div className="popup-content">
              <h4>{selectedResult.name}</h4>
              <p>
                Coordinates: {selectedResult.coordinates[1].toFixed(4)}, {selectedResult.coordinates[0].toFixed(4)}
              </p>
              {selectedResult.properties && (
                <div className="popup-properties">
                  {Object.entries(selectedResult.properties).map(([key, value]) => (
                    <p key={key}>
                      <strong>{key}:</strong> {value}
                    </p>
                  ))}
                </div>
              )}
            </div>
          </Popup>
        )}
      </Map>
    </div>
  );
}

export default GeocodingMap;
```

## Advanced Features

### 1. Coordinate Search
Add support for coordinate-based searches:

```jsx
// Add to geocodingService.js
parseCoordinates(query) {
  const coordPatterns = [
    /^(-?\d+\.?\d*),?\s*(-?\d+\.?\d*)$/, // "lat,lng" or "lat lng"
    /^(-?\d+\.?\d*)[,\s]+(-?\d+\.?\d*)$/, // Various separators
  ];

  for (const pattern of coordPatterns) {
    const match = query.match(pattern);
    if (match) {
      const lat = parseFloat(match[1]);
      const lng = parseFloat(match[2]);
      
      if (lat >= -90 && lat <= 90 && lng >= -180 && lng <= 180) {
        return { coordinates: [lng, lat], isCoordinate: true };
      }
    }
  }
  
  return null;
}
```

## Best Practices

### 1. Performance Optimization
- Implement proper debouncing for search inputs
- Use React.memo for components that don't need frequent re-renders
- Implement virtual scrolling for large result sets
- Cache search results to avoid redundant API calls

### 2. Error Handling
- Implement retry logic for failed requests
- Show user-friendly error messages
- Handle network connectivity issues
- Implement fallback geocoding services

### 3. Accessibility
- Add proper ARIA labels and roles
- Ensure keyboard navigation works
- Provide screen reader support
- Use semantic HTML elements

### 4. Mobile Responsiveness
- Implement touch-friendly interfaces
- Handle different screen sizes
- Consider mobile-specific interactions
- Optimize for slower network connections

### 5. Security
- Validate and sanitize user inputs
- Implement rate limiting
- Use HTTPS for all API calls
- Keep API keys secure (use environment variables)

### 6. Testing
